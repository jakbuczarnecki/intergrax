def blob_stats(path: str):
    r = git("cat-file", "-t", f"{BASELINE_SHA}:{path}")
    if r.returncode != 0:
        return {"path": path, "exists": False}
    kind = r.stdout.strip()
    if kind == "tree":
        ls = git("ls-tree", "-r", "--name-only", BASELINE_SHA, path + "/").stdout.split("\n")
        files = [x for x in ls if x]
        return {"path": path, "exists": True, "kind": "tree", "n_files": len(files)}
    b = show_bytes(path)
    return {"path": path, "exists": True, "kind": "blob", "lines": b.count(b"\n") + (0 if b.endswith(b"\n") else 1),
            "blob_sha256": sha256_bytes(b), "git_blob": git("rev-parse", f"{BASELINE_SHA}:{path}").stdout.strip()}


def resolve_link(base_dir: str, target: str) -> str:
    parts = []
    for seg in (base_dir + "/" + target).replace("\\", "/").split("/"):
        if seg == "..":
            if parts:
                parts.pop()
        elif seg and seg != ".":
            parts.append(seg)
    return "/".join(parts)
