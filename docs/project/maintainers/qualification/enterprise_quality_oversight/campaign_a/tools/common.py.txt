def header_block(extra: str = "") -> str:
    reg = WS / "SOURCE_REGISTRY.json"
    digest = sha256_file(reg) if reg.exists() else "pending"
    return (f"<!-- Campaign: {CAMPAIGN} | BASELINE_SHA: {BASELINE_SHA} | generated_utc: {utc_now()} | "
            f"source_registry_sha256: {digest} -->\n")


def utc_now() -> str:
    import datetime
    return datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")


def sha256_file(p) -> str:
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def sha256_bytes(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


def git(*args: str, cwd: Path = WT, check: bool = False) -> subprocess.CompletedProcess:
    # read-only git only
    allowed = {"log", "show", "diff", "rev-parse", "merge-base", "cat-file", "ls-tree", "blame", "grep", "status",
               "rev-list", "name-rev", "branch", "for-each-ref", "ls-files", "describe", "cherry", "shortlog",
               "diff-tree", "show-ref"}
    if args[0] not in allowed:
        raise RuntimeError(f"git subcommand not allowed: {args[0]}")
    r = subprocess.run(["git", "-C", str(cwd), *args], capture_output=True, text=True, encoding="utf-8", errors="replace")
    if check and r.returncode != 0:
        raise RuntimeError(r.stderr)
    return r


def show_text(path: str, sha: str = BASELINE_SHA) -> str | None:
    r = subprocess.run(["git", "-C", str(WT), "show", f"{sha}:{path}"], capture_output=True)
    if r.returncode != 0:

    return r.stdout.decode("utf-8", errors="replace")

def path_exists_at(path: str, sha: str = BASELINE_SHA) -> bool:
    return git("cat-file", "-e", f"{sha}:{path}").returncode == 0


def commit_exists(sha: str) -> bool:
    return git("cat-file", "-e", f"{sha}^{{commit}}").returncode == 0


def is_ancestor(sha: str, of: str = BASELINE_SHA) -> bool:
    return git("merge-base", "--is-ancestor", sha, of).returncode == 0


def write_json(path, obj) -> None:
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    Path(path).write_text(json.dumps(obj, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")


def write_jsonl(path, rows) -> None:
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    with open(path, "w", encoding="utf-8", newline="\n") as f:
        for r in rows:
            f.write(json.dumps(r, ensure_ascii=False, sort_keys=False) + "\n")


def read_jsonl(path):
    out = []
    p = Path(path)
    if not p.exists():
        return out
    with open(p, encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if line:
                out.append(json.loads(line))
    return out


def read_json(path):
    return json.loads(Path(path).read_text(encoding="utf-8"))


def clean_cell(s: str) -> str:
    s = s.strip()
    s = re.sub(r"\*\*", "", s)
    s = s.replace("`", "")
    return s.strip()


def split_row(line: str) -> list[str]:
    """Split a markdown table row on unescaped pipes outside inline code."""
    s = line.strip()
    if s.startswith("|"):
        s = s[1:]
    if s.endswith("|"):
        s = s[:-1]
    cells, cur, in_code, i = [], [], False, 0
    while i < len(s):
        ch = s[i]
        if ch == "\\" and i + 1 < len(s) and s[i + 1] == "|":
            cur.append("|")
            i += 2
            continue
        if ch == "`":
            in_code = not in_code
        if ch == "|" and not in_code:
            cells.append("".join(cur))
            cur = []
        else:
            cur.append(ch)
        i += 1
    cells.append("".join(cur))
    return [c.strip() for c in cells]


def log_read(path: str, sha: str = BASELINE_SHA, lines: str = "all", note: str = "") -> None:
    """Append to READ_LOG.jsonl (path, blob sha, line range)."""
    r = git("rev-parse", f"{sha}:{path}")
    blob = r.stdout.strip() if r.returncode == 0 else "MISSING"
    with open(WS / "READ_LOG.jsonl", "a", encoding="utf-8", newline="\n") as f:
        f.write(json.dumps({"path": path, "sha": sha, "blob": blob, "lines": lines, "note": note,
                            "utc": utc_now()}, ensure_ascii=False) + "\n")
