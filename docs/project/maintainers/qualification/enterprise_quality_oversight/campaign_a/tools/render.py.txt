Roll-up numbers carry <!--rollup:KEY=VALUE--> markers that check C08 recomputes independently (rollup_calc)."""

def esc(s, n=None):
    s = "" if s is None else str(s)
    s = s.replace("|", "\\|").replace("\n", " ").replace("\r", " ")
    return (s[: n - 1] + "…") if n and len(s) > n else s


def roll(key: str) -> str:
    """number with its rollup marker"""
    return f"{compute(key)}<!--rollup:{key}={compute(key)}-->"


def findings_by_scope():
    m = defaultdict(list)
    for f in fl.consolidate():
        if f.get("status", "OPEN") != "OPEN":
            continue
        for s in f.get("scope", []):
            m[s].append(f["finding_id"])
    return m


def head(title: str, intro: str = "") -> list[str]:
    return [header_block().rstrip("\n"), "", f"# {title}", "", "> Generated by `tools/render.py` from `data/`. EQO-A is a read-only baseline audit; this document is **not** a certificate and has **not** been independently accepted until the auditor report says so.", "", intro, ""]


def table(headers, rows):
    out = ["| " + " | ".join(headers) + " |", "|" + "|".join("---" for _ in headers) + "|"]
    for r in rows:
        out.append("| " + " | ".join(esc(c) for c in r) + " |")
    return out
