def tables(text: str):
    out, cur, hdr = [], None, None
    for ln, l in enumerate(text.split("\n"), 1):
        if l.startswith("|"):
            c = [clean_cell(x) for x in split_row(l)]
            if cur is None:
                cur, hdr = [], c
                continue
            if all(set(x) <= set("-: ") for x in c):
                continue
            cur.append({"line": ln, **{(hdr[i] if i < len(hdr) and hdr[i] else f"c{i}"): c[i] for i in range(len(c))}})
        else:
            if cur is not None:
                out.append({"header": hdr, "rows": cur})
            cur, hdr = None, None
    if cur is not None:
        out.append({"header": hdr, "rows": cur})
    return out


def main():
    summary = {}
    for name, path in REGS.items():
        t = show_text(path)
        if t is None:
            summary[name] = {"path": path, "exists": False}
            continue
        log_read(path, note="A7-T01 parse_registers")
        tabs = tables(t)
        write_json(WS / "extracts" / "registers" / f"{name}.json", {"path": path, "tables": tabs, "lines": t.count("\n") + 1})
        summary[name] = {"path": path, "exists": True, "lines": t.count("\n") + 1, "tables": len(tabs), "rows": sum(len(x["rows"]) for x in tabs)}
    write_json(WS / "extracts" / "registers_summary.json", summary)
    for k, v in summary.items():
        print(k, v.get("exists"), v.get("rows"), v.get("lines"))


if __name__ == "__main__":
    main()
